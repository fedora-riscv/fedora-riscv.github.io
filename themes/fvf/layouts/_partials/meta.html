{{- /* Display metadata for a page: data/docs.yaml first, then front matter. Returns a dict. */ -}} {{- $en := eq .Language.Lang "en" }} {{- $key := "" }}{{ with .File }}{{ $key = .ContentBaseName }}{{ end }} {{- $d := index hugo.Data.docs.pages $key | default dict }} {{- $sec := index hugo.Data.docs.sections .Section | default dict }} {{- $title := .Title }} {{- if .IsSection }}{{ $title = (cond $en $sec.title_en $sec.title) | default .Title }}{{ end }} {{- with (cond $en $d.title_en $d.title) }}{{ $title = . }}{{ end }} {{- $summary := strings.TrimSpace (.Params.summary | default "") }} {{- with (cond $en $d.summary_en $d.summary) }}{{ $summary = . }}{{ end }} {{- if and (not $summary) .IsSection }}{{ $summary = (cond $en $sec.description_en $sec.description) | default "" }}{{ end }} {{- if and (not $summary) .IsPage }}{{ $summary = .Plain | strings.TrimSpace | replaceRE `\s+` " " | replaceRE `^\[TOC\]\s*` "" | truncate 90 }}{{ end }} {{- $subtitle := "" }} {{- with findRESubmatch `]*size=['"]?20['"]?[^>]*>([^<]+)` .RawContent 1 }}{{ $subtitle = index (index . 0) 1 | strings.TrimSpace }}{{ end }} {{- with .Params.subtitle }}{{ $subtitle = . }}{{ end }} {{- $edit := "" }} {{- with .File }} {{- if $en }}{{ $edit = printf "%s/blob/main/content-en/%s" site.Params.siteRepo .Path }} {{- else if $sec.edit }}{{ $edit = printf "%s%s" $sec.edit (strings.TrimPrefix (printf "%s/" $.Section) .Path) }} {{- else }}{{ $edit = printf "%s/blob/main/content/%s" site.Params.siteRepo .Path }}{{ end }} {{- end }} {{- $seriesTitle := "" }}{{ with $d.series }}{{ with index hugo.Data.docs.series . }}{{ $seriesTitle = (cond $en .title_en .title) | default .title }}{{ end }}{{ end }} {{- return dict "title" $title "summary" $summary "subtitle" $subtitle "n" ($d.n | default "") "series" ($d.series | default "") "seriesTitle" $seriesTitle "kind" ((cond $en $d.kind_en $d.kind) | default "") "edit" $edit }}
